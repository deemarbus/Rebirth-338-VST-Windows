cmake_minimum_required(VERSION 3.25)
project(rebirth_bridge_worker LANGUAGES C)
include(CTest)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Build the backend as Windows x86, for example Visual Studio -A Win32")
endif()
if(BUILD_TESTING)
    add_executable(worker-run-test worker-run-test.c)
    target_compile_features(worker-run-test PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(worker-run-test PRIVATE /W4 /std:c11)
        set_property(TARGET worker-run-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
    add_test(NAME worker-linear-transport COMMAND worker-run-test)
endif()
add_executable(rebirth-bridge-worker bridge-worker.c)
target_compile_features(rebirth-bridge-worker PRIVATE c_std_11)
target_link_libraries(rebirth-bridge-worker PRIVATE advapi32)
if(MSVC)
    set_property(TARGET rebirth-bridge-worker PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(rebirth-bridge-worker PRIVATE /W4 /std:c11)
else()
    target_compile_options(rebirth-bridge-worker PRIVATE -Wall -Wextra -Werror)
    target_link_options(rebirth-bridge-worker PRIVATE -municode -static)
    target_link_libraries(rebirth-bridge-worker PRIVATE m)
endif()
