cmake_minimum_required(VERSION 3.25)
project(ReBirthBridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the plug-in for Windows x64; the engine worker is a separate x86 target.")
endif()
set(VST3_SDK_ROOT "" CACHE PATH "Unmodified official VST3 SDK 3.8.1 checkout")
if(NOT EXISTS "${VST3_SDK_ROOT}/public.sdk/source/vst/vstaudioeffect.h")
    message(FATAL_ERROR "Set VST3_SDK_ROOT to the initialized official SDK checkout.")
endif()
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ADD_VST3_UTILITIES OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory("${VST3_SDK_ROOT}" sdk)

include(CTest)
set(RB_BUILD_TESTING ${BUILD_TESTING})
set(BUILD_TESTING OFF)
set(LIBSAMPLERATE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LIBSAMPLERATE_INSTALL OFF CACHE BOOL "" FORCE)
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()
include(FetchContent)
FetchContent_Declare(libsamplerate
    URL https://codeload.github.com/libsndfile/libsamplerate/zip/refs/tags/0.2.2
    URL_HASH SHA256=7BD06FABD57027E9C0FA22E6BD873E4916AB96EA132D6CB1B76E7FDD2ADE1D20
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(libsamplerate)
set(BUILD_TESTING ${RB_BUILD_TESTING})


smtg_add_vst3plugin(ReBirthBridge processor.cpp controller.cpp editor.cpp factory.cpp)
target_include_directories(ReBirthBridge PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(ReBirthBridge PRIVATE sdk samplerate comctl32 user32 gdi32)
target_link_libraries(ReBirthBridge PRIVATE sdk samplerate comctl32 user32 gdi32)
if(MSVC)
    target_compile_options(ReBirthBridge PRIVATE /W4 /permissive-)
endif()
if(BUILD_TESTING)
    add_executable(bridge-adapter-test tests/adapter-test.cpp)
    target_link_libraries(bridge-adapter-test PRIVATE samplerate)
    target_compile_features(bridge-adapter-test PRIVATE cxx_std_17)
    target_link_libraries(bridge-adapter-test PRIVATE samplerate)
    add_test(NAME bridge-adapter COMMAND bridge-adapter-test)
endif()
